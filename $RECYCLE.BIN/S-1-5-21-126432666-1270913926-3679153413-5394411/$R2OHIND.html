import PipelinePollerWorker from "../modules/pipelinepoller.module.js";
import AnsiblePollerWorker from "../modules/ansiblepoller.module.js";
 
/**
 * Controller to fetch all polling jobs from the database
 * Supports pagination and status filtering for both pipeline and ansible pollers
 */
export const getAllPollerJobs = async (req, res) => {
  try {
    console.log("Getting polling jobs...");
    console.log("Query params:", req.query);
    console.log("User from request:", req.user?.email || 'No user'); 
    
    const page = parseInt(req.query.page) || 1;
    const limit = parseInt(req.query.limit) || 10;
    const status = req.query.status;
    const type = req.query.type; // 'pipeline', 'ansible', or 'all'
    const sortBy = req.query.sortBy || 'createdAt';
    const sortOrder = req.query.sortOrder || 'desc';
    const skip = (page - 1) * limit;
 
    console.log(`📊 Page: ${page}, Limit: ${limit}, Status filter: ${status}, Worker type filter: ${type}, Sort: ${sortBy} ${sortOrder}`);
 
    // Build filter object
    const filter = {};
    if (status) {
      filter.status = status.toLowerCase(); // Match enum values
    }
 
    let allJobs = [];
    let totalJobs = 0;
 
    try {
      // Fetch based on type filter
      if (type === 'pipeline' || !type || type === 'all') {
        console.log("Fetching pipeline jobs...");
        const pipelineJobs = await PipelinePollerWorker.find(filter)
          .sort({ createdAt: -1 })
          .lean();
        
        console.log(`Found ${pipelineJobs.length} pipeline jobs`);
        
        // Add job type identifier
        const pipelineJobsWithType = pipelineJobs.map(job => ({
          ...job,
          workerType: 'pipeline',
          jobName: `Pipeline ${job.jobType} - ${job.pipelineId}`,
          uniqueId: job.pipelineId
        }));
        
        allJobs = [...allJobs, ...pipelineJobsWithType];
      }
 
      if (type === 'ansible' || !type || type === 'all') {
        console.log("Fetching ansible jobs...");
        const ansibleJobs = await AnsiblePollerWorker.find(filter)
          .sort({ createdAt: -1 })
          .lean();
        
        console.log(`Found ${ansibleJobs.length} ansible jobs`);
        
        // Add job type identifier
        const ansibleJobsWithType = ansibleJobs.map(job => ({
          ...job,
          workerType: 'ansible',
          jobName: `Ansible ${job.jobType} - ${job.jobId}`,
          uniqueId: job.jobId
        }));
        
        allJobs = [...allJobs, ...ansibleJobsWithType];
      }
    } catch (dbError) {
      console.error("Database query error:", dbError);
      // Return empty results if database query fails
      allJobs = [];
    }
 
    console.log(`Total jobs found: ${allJobs.length}`);
 
    // Sort all jobs based on provided parameters
    allJobs.sort((a, b) => {
      const direction = sortOrder === 'desc' ? -1 : 1;
      
      switch (sortBy) {
        case 'jobName':
          return direction * (a.jobName || '').localeCompare(b.jobName || '');
        case 'workerType':
          return direction * (a.workerType || '').localeCompare(b.workerType || '');
        case 'jobType':
          return direction * (a.jobType || '').localeCompare(b.jobType || '');
        case 'environment':
          return direction * (a.envId || '').localeCompare(b.envId || '');
        case 'owner':
          return direction * (a.owner || '').localeCompare(b.owner || '');
        case 'status':
          return direction * (a.status || '').localeCompare(b.status || '');
        case 'createdAt':
        default:
          return direction * (new Date(b.createdAt) - new Date(a.createdAt));
      }
    });
 
    // Calculate total count
    totalJobs = allJobs.length;
 
    // Apply pagination after sorting
    const paginatedJobs = allJobs.slice(skip, skip + limit);
 
    const totalPages = Math.ceil(totalJobs / limit) || 1;
 
    console.log(`📬 Returning ${paginatedJobs.length} poller jobs (${totalJobs} total)`);
 
    // Transform data to match frontend expectations
    const transformedJobs = paginatedJobs.map(job => ({
      _id: job._id,
      jobName: job.jobName,
      workerType: job.workerType,
      uniqueId: job.uniqueId,
      envId: job.envId,
      envType: job.envType,
      tenantId: job.tenantId,
      owner: job.owner,
      jobType: job.jobType,
      status: job.status,
      pipelineId: job.pipelineId || null,
      jobId: job.jobId || null,
      exceptionCount: job.exceptionCount || 0,
      error: job.error || null,
      createdAt: job.createdAt,
      updatedAt: job.updatedAt
    }));
 
    res.status(200).json({
      success: true,
      jobs: transformedJobs,
      pagination: {
        currentPage: page,
        totalPages,
        totalJobs,
        hasNextPage: page < totalPages,
        hasPreviousPage: page > 1
      },
      totalJobs
    });
 
  } catch (error) {
    console.error("❌ Error fetching polling jobs:", error);
    res.status(500).json({
      success: false,
      message: "Failed to fetch polling jobs",
      error: process.env.NODE_ENV === 'development' ? error.message : 'Internal server error'
    });
  }
};
