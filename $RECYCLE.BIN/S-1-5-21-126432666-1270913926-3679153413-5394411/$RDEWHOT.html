import productIdValidationService from '../services/productIdvalidation.service.js';
 
/**
 * Middleware to validate product IDs in request body
 * Can be used in environment creation routes or other endpoints that need product ID validation
 */
export const validateProductIdsMiddleware = (options = {}) => {
    const {
        fieldName = 'productIds',     // Field name in request body that contains product IDs
        branchField = 'tcxManifestVersion', // Field name for branch in request body
        branch = null,                // Static branch to validate against (optional)
        required = true,              // Whether product IDs are required
        allowEmpty = false            // Whether to allow empty arrays
    } = options;
 
    return async (req, res, next) => {
        try {
            const productIds = req.body[fieldName];
            // Get branch from request body or use provided branch
            const branchToUse = branch || req.body[branchField] || req.query.branch;
            
            console.log(`[Middleware] Branch source: ${branch ? 'static' : 'request'}, value: "${branchToUse}"`);
            
            if (!branchToUse) {
                return res.status(400).json({
                    success: false,
                    error: `Branch information is required (provide '${branchField}' in request body or 'branch' in query)`,
                    field: branchField
                });
            }
 
            // Check if product IDs are provided when required
            if (required && !productIds) {
                return res.status(400).json({
                    success: false,
                    error: `${fieldName} is required`,
                    field: fieldName
                });
            }
 
            // If not provided and not required, skip validation
            if (!productIds && !required) {
                return next();
            }
 
            // Validate that it's an array
            if (!Array.isArray(productIds)) {
                return res.status(400).json({
                    success: false,
                    error: `${fieldName} must be an array`,
                    field: fieldName
                });
            }
 
            // Check if empty array is allowed
            if (productIds.length === 0 && !allowEmpty) {
                return res.status(400).json({
                    success: false,
                    error: `${fieldName} cannot be empty`,
                    field: fieldName
                });
            }
 
            // If empty but allowed, skip validation
            if (productIds.length === 0 && allowEmpty) {
                return next();
            }
 
            console.log(`[Middleware] Validating product IDs in field '${fieldName}':`, productIds);
            console.log(`[Middleware] Using branch: "${branchToUse}"`);
 
            // Validate the product IDs
            const validationResult = await productIdValidationService.validateProductIds(productIds, branchToUse);
 
            if (!validationResult.isValid) {
                return res.status(400).json({
                    success: false,
                    error: validationResult.errorMessage,
                    invalidProductIds: validationResult.invalidProductIds || [],
                    mutuallyExclusiveConflicts: validationResult.mutuallyExclusiveConflicts || [],
                    field: fieldName
                });
            }
 
            // Add validation info to request for use in route handlers
            req.productIdValidation = {
                isValid: true,
                validatedProductIds: productIds,
                field: fieldName,
                branch: branchToUse
            };
 
            console.log(`[Middleware] Product ID validation successful for field '${fieldName}'`);
            next();
 
        } catch (error) {
            console.error(`[Middleware] Product ID validation error for field '${fieldName}':`, error);
            res.status(500).json({
                success: false,
                error: 'Internal server error during product ID validation',
                details: error.message,
                field: fieldName
            });
        }
    };
};
 
/**
 * Middleware to validate product IDs in query parameters
 * Useful for GET requests that include product IDs as comma-separated values
 */
export const validateProductIdsQueryMiddleware = (options = {}) => {
    const {
        paramName = 'productIds',  // Query parameter name
        branch = null,             // Static GitLab branch to validate against (optional)
        required = true,           // Whether the parameter is required
        separator = ','            // Separator for multiple product IDs
    } = options;
 
    return async (req, res, next) => {
        try {
            const productIdsParam = req.query[paramName];
            // Get branch from query or use provided branch
            const branchToUse = branch || req.query.branch || req.body?.tcxManifestVersion;
            
            console.log(`[Middleware Query] Branch source: ${branch ? 'static' : 'request'}, value: "${branchToUse}"`);
            
            if (!branchToUse) {
                return res.status(400).json({
                    success: false,
                    error: 'Branch information is required (provide branch in query parameters)',
                    parameter: 'branch'
                });
            }
 
            // Check if product IDs are provided when required
            if (required && !productIdsParam) {
                return res.status(400).json({
                    success: false,
                    error: `${paramName} query parameter is required`,
                    parameter: paramName
                });
            }
 
            // If not provided and not required, skip validation
            if (!productIdsParam && !required) {
                return next();
            }
 
            // Parse comma-separated product IDs into array
            const productIds = productIdsParam.split(separator).map(id => id.trim()).filter(id => id);
 
            if (productIds.length === 0) {
                return res.status(400).json({
                    success: false,
                    error: `${paramName} cannot be empty`,
                    parameter: paramName
                });
            }
 
            console.log(`[Middleware] Validating product IDs in query '${paramName}':`, productIds);
            console.log(`[Middleware Query] Using branch: "${branchToUse}"`);
 
            // Validate the product IDs
            const validationResult = await productIdValidationService.validateProductIds(productIds, branchToUse);
 
            if (!validationResult.isValid) {
                return res.status(400).json({
                    success: false,
                    error: validationResult.errorMessage,
                    invalidProductIds: validationResult.invalidProductIds || [],
                    mutuallyExclusiveConflicts: validationResult.mutuallyExclusiveConflicts || [],
                    parameter: paramName
                });
            }
 
            // Add validation info to request for use in route handlers
            req.productIdValidation = {
                isValid: true,
                validatedProductIds: productIds,
                parameter: paramName,
                branch: branchToUse
            };
 
            console.log(`[Middleware] Product ID validation successful for query '${paramName}'`);
            next();
 
        } catch (error) {
            console.error(`[Middleware] Product ID validation error for query '${paramName}':`, error);
            res.status(500).json({
                success: false,
                error: 'Internal server error during product ID validation',
                details: error.message,
                parameter: paramName
            });
        }
    };
};
 
/**
 * Helper function to create a validation middleware with specific configuration
 * @param {Object} config - Configuration for validation
 * @returns {Function} Configured middleware function
 */
export const createProductIdValidator = (config) => {
    return validateProductIdsMiddleware(config);
};

export const validateProductIdsQuery = validateProductIdsQueryMiddleware({
    paramName: 'productIds',
    required: true
