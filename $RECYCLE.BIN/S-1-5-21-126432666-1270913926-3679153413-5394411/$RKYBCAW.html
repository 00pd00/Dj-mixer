/**

 * @returns {{streamId: string}}
 */
export function parseTenantYaml(yamlContent) {
  const GITLAB_STREAM_ID_FIELD_NAME = "STREAM_ID";
  const parsed = yaml.load(yamlContent);
  const streamId = parsed.variables[GITLAB_STREAM_ID_FIELD_NAME];
