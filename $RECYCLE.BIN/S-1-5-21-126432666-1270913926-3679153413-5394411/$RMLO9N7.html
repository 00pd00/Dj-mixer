const GITLAB_API_URL = process.env.GITLAB_API_URL;
const GITLAB_TOKEN = process.env.GITLAB_TOKEN;
const GITLAB_PROJECT_ID = process.env.GITLAB_PROJECT_ID || 49317;

/**
 * Creates a new GitLab issue for an environment request.
 * Uses product names for the issue title and links the created issue to the environment.
 *
 * @param {Object} params - Parameters for issue creation
 * @param {string} params.environmentId - The environment MongoDB document ID
 * @param {Array<string>} params.productIds - Array of product MongoDB IDs
 * @param {string} [params.description] - Optional description for the issue
 * @returns {Promise<Object>} The created GitLab issue data
 */
 
export const createGitlabIssue = async ({ environmentId, productIds, requestedBy, description }) => {
  try {
    // Fetch the products to get their names instead of just IDs
    const products = await Product.find({ productIds: { $in: productIds } });
    // Get product names for title and description
    const productNames = products.map(p => p.name);
    let title;
    if (products.length === 1) {
      title = `Environment requested by ${requestedBy} for ${productNames[0]}`;
    } else{
      // For 2-3 products, list their names
      title = `Environment requested by ${requestedBy}`;
     }
    // Add product names section to description
    let productNamesSection = "\n\n### Products Name\n";
    productNames.forEach((name, index) => {
      productNamesSection += `Product ${index + 1}: ${name}\n\n`;
    });
    
    // Insert product names section before "Selected Products"
    let updatedDescription = description || "Environment requested via onboarding portal.";
    if (description && description.includes("### Selected Products")) {
      updatedDescription = description.replace("### Selected Products", `${productNamesSection}### Selected Products`);
    } else {
      updatedDescription = description + productNamesSection;
    }
    
    const assignee_id = 3652;
    const labels = "environment::requested";
 
    // Create the GitLab issue
    const response = await axios.post(
      `${GITLAB_API_URL}/projects/${GITLAB_PROJECT_ID}/issues`,
      {
        title,
        description: updatedDescription,
        assignee_ids: [assignee_id],
        labels,
      },
      {
        headers: {
          "Content-Type": "application/json",
          "Authorization": `Bearer ${GITLAB_TOKEN}`,
        },
      }
    );
 
    const issueId = response.data.iid;
 
    // Update environment with the issue ID
    if (environmentId && issueId) {
      await Environment.findByIdAndUpdate(
        environmentId,
        { gitlabIssueId: issueId },
        { new: true }
      );
    }
 
    return response.data;
  } catch (error) {
    console.error("Error creating GitLab issue:", error);
    throw error;
  }
};

/**
 * Updates the labels of a GitLab issue.
 * Removes the old label and adds the new label to the issue.
 *
 * @param {number|string} issueId - The GitLab issue IID
 * @param {string} newLabel - The label to add
 * @param {string} oldLabel - The label to remove
 */
 
export const updateGitlabIssueLabel = async (issueId, newLabel, oldLabel) => {
 
  // Get current labels
  const issueRes = await axios.get(
    `${GITLAB_API_URL}/projects/${GITLAB_PROJECT_ID}/issues/${issueId}`,
    {
      headers: {
        "Authorization": `Bearer ${GITLAB_TOKEN}`,
      },
    }
  );
  let labels = issueRes.data.labels || [];
  labels = labels.filter(l => l !== oldLabel);
  if (!labels.includes(newLabel)) labels.push(newLabel);
 
  // Update labels
  await axios.put(
    `${GITLAB_API_URL}/projects/${GITLAB_PROJECT_ID}/issues/${issueId}`,
    { labels },
    {
      headers: {
        "Content-Type": "application/json",
        "Authorization": `Bearer ${GITLAB_TOKEN}`,
      },
    }
  );
};
