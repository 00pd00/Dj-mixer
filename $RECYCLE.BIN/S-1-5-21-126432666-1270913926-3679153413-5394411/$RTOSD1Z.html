import TCXSecretsManager from "../SecretReader.js";
import AWSCredentialManager from "../Creds/AWSCredsManager.js";
import Environment from "../modules/environment.module.js";
import configs from "../data/configs.json" with { type: "json" };
 
/**
 * Custom error classes for better error handling and service layer separation
 */
export class NotFoundError extends Error {
    constructor(message) {
        super(message);
        this.name = 'NotFoundError';
    }
}
 
export class BadRequestError extends Error {
    constructor(message) {
        super(message);
        this.name = 'BadRequestError';
    }
}
 
export class VaultError extends Error {
    constructor(message, details) {
        super(message);
        this.name = 'VaultError';
        this.details = details;
    }
}
 
export class ConfigurationError extends Error {
    constructor(message) {
        super(message);
        this.name = 'ConfigurationError';
    }
}
 
/**
 * Constants from configuration
 */
const VAULT_PATHS = configs.SECRETS.VAULT_PATHS;
const FIELD_NAMES = configs.SECRETS.FIELD_NAMES;
const DEMO_PASSWORDS = configs.SECRETS.DEMO_PASSWORDS;
const ENVIRONMENT_STATUS = {
    READY: configs.ENVIRONMENT_STATUSES.READY,
    FAILED: configs.ENVIRONMENT_STATUSES.FAILED
};
 
/**
 * Check if bypass mode is enabled
 * @returns {boolean} True if bypass mode is enabled
 */
const isInBypassMode = () => {
    return process.env.BYPASS_SECRETS_DISPLAY === 'true';
};
 
/**
 * Get demo passwords for bypass mode
 * @returns {Object} Demo password object
 */
const getDemoPasswords = () => ({ ...DEMO_PASSWORDS });
 
/**
 * Validate that all required configuration is present
 * @throws {ConfigurationError} If required configuration is missing
 */
const validateConfiguration = () => {
    const awsAccountId = process.env.AWS_ACCOUNT_ID;
    if (!awsAccountId) {
        throw new ConfigurationError("AWS_ACCOUNT_ID environment variable not configured");
    }
};
 
/**
 * Validate environment input
 * @param {string} envId - Environment ID
 * @throws {BadRequestError} If environment ID is invalid
 */
const validateEnvironmentId = (envId) => {
    if (!envId || typeof envId !== 'string' || envId.trim() === '') {
        throw new BadRequestError("Environment ID is required and must be a valid string");
    }
};
 
/**
 * Find and validate environment
 * @param {string} envId - Environment ID
 * @returns {Promise<Object>} Environment object
 * @throws {NotFoundError} If environment not found
 * @throws {BadRequestError} If environment status is not Ready or Failed
 */
const findAndValidateEnvironment = async (envId) => {
    const environment = await Environment.findById(envId);
    if (!environment) {
        throw new NotFoundError(`Environment with ID ${envId} not found`);
    }
 
    // Allow password access for both Ready and Failed environments
    if (environment.status !== ENVIRONMENT_STATUS.READY && environment.status !== ENVIRONMENT_STATUS.FAILED) {
        throw new BadRequestError(
            `Environment passwords are only available when status is Ready or Failed. Current status: ${environment.status}`
        );
    }
 
    return environment;
};
 
/**
 * Check if passwords are already stored in database
 * @param {Object} environment - Environment object
 * @returns {boolean} True if passwords exist
 */
const hasStoredPasswords = (environment) => {
    return !!(environment.tcx_user_password && environment.dc_server_password);
};
 
/**
 * Create password response object
 * @param {Object} passwords - Password data
 * @param {Object} environment - Environment object
 * @returns {Object} Formatted password response
 */
const createPasswordResponse = (passwords, environment) => {
    return {
        tcx_user_password: passwords.tcx_user_password || passwords.tcxUserPassword,
        dc_server_password: passwords.dc_server_password || passwords.dcServerPassword,
        tenantId: environment.tenantId,
        environmentType: environment.environmentType
    };
};
 
/**
 * Fetch passwords from Vault
 * @param {Object} environment - Environment object
 * @returns {Promise<Object>} Password data from Vault
 * @throws {VaultError} If Vault operations fail
 */
const fetchPasswordsFromVault = async (environment) => {
    const { tenantId, environmentType, awsRegion } = environment;
    const awsAccountId = process.env.AWS_ACCOUNT_ID;
    
    // Get duration from configuration for password display use case
    const durationSeconds = configs.SECRETS.PASSWORD_DISPLAY_DURATION_SECONDS;
 
    console.log(`[SecretsService] Fetching passwords from Vault for ${tenantId}/${environmentType} with ${durationSeconds}s credentials (from config)`);
 
    try {
        // Step 1: Get temporary credentials from AWSCredentialManager (for the chained base roles)
        console.log(`[SecretsService] Step 1: Getting temporary AWS credentials for ${durationSeconds} seconds`);
        const awsCredManager = new AWSCredentialManager();
        const baseCredentials = await awsCredManager.getChainedRoleCredentials(durationSeconds);
        
        // Step 2: Use base credentials to assume the tenant-specific role
        console.log(`[SecretsService] Step 2: Using base credentials to assume tenant-specific role`);
        console.log(`[SecretsService] Base credentials type:`, typeof baseCredentials);
        console.log(`[SecretsService] Base credentials keys:`, Object.keys(baseCredentials || {}));
        if (baseCredentials) {
            console.log(`[SecretsService] Base credentials AccessKeyId:`, baseCredentials.AccessKeyId?.substring(0, 8) + '...');
            console.log(`[SecretsService] Base credentials has SessionToken:`, !!baseCredentials.SessionToken);
        }
        
        const tenantRoleArn = `arn:aws:iam::${awsAccountId}:role/VaultOperatorReadAccessRole-${tenantId}`;
        console.log(`[SecretsService] Assuming tenant role: ${tenantRoleArn}`);
        console.log(`[SecretsService] Duration for tenant role assumption: ${durationSeconds} seconds`);
        
        const tenantCredentials = await awsCredManager.assumeRole(
            tenantRoleArn,
            baseCredentials,
            `TenantVaultSession-${Date.now()}`,
            durationSeconds
        );
        
        console.log(`[SecretsService] Tenant credentials received:`, !!tenantCredentials);
        if (tenantCredentials) {
            console.log(`[SecretsService] Tenant credentials AccessKeyId:`, tenantCredentials.AccessKeyId?.substring(0, 8) + '...');
            console.log(`[SecretsService] Tenant credentials has SessionToken:`, !!tenantCredentials.SessionToken);
        }
        
        // Step 3: Use tenant credentials with SecretReader
        console.log(`[SecretsService] Step 3: Using tenant-specific credentials with SecretReader`);
        console.log(`[SecretsService] Creating TCXSecretsManager with parameters:`);
        console.log(`[SecretsService] - tenantId: ${tenantId}`);
        console.log(`[SecretsService] - environmentType: ${environmentType}`);
        console.log(`[SecretsService] - awsAccountId: ${awsAccountId}`);
        console.log(`[SecretsService] - awsRegion: ${awsRegion || "us-east-1"}`);
        console.log(`[SecretsService] - durationSeconds: ${durationSeconds}`);
        console.log(`[SecretsService] - tenantCredentials passed:`, !!tenantCredentials);
        if (tenantCredentials) {
            console.log(`[SecretsService] - tenantCredentials AccessKeyId: ${tenantCredentials.AccessKeyId?.substring(0, 8)}...`);
            console.log(`[SecretsService] - tenantCredentials structure:`, Object.keys(tenantCredentials));
        }
        
        const secretsManager = new TCXSecretsManager(
            tenantId,
            environmentType,
            awsAccountId,
            awsRegion || "us-east-1",
            tenantCredentials,
            durationSeconds
        );
 
        // Step 4: Authenticate with Vault using the tenant credentials
        console.log(`[SecretsService] Step 4: Authenticating with Vault using tenant credentials`);
        console.log(`[SecretsService] About to call secretsManager.getAssumedCreds()`);
        
        const awsCredentials = await secretsManager.getAssumedCreds();
        
        console.log(`[SecretsService] AWS credentials from SecretReader:`, !!awsCredentials);
        if (awsCredentials) {
            console.log(`[SecretsService] SecretReader credentials AccessKeyId:`, awsCredentials.AccessKeyId?.substring(0, 8) + '...');
            console.log(`[SecretsService] SecretReader credentials has SessionToken:`, !!awsCredentials.SessionToken);
            console.log(`[SecretsService] SecretReader credentials structure:`, Object.keys(awsCredentials));
        } else {
            console.error(`[SecretsService] No credentials returned from SecretReader!`);
        }
        
        const payload = secretsManager.getSignedIamPayload(awsCredentials);
        console.log(`[SecretsService] Generated IAM payload for Vault login`);
        
        const vaultToken = await secretsManager.vaultLogin(payload);
        console.log(`[SecretsService] Successfully obtained Vault token:`, !!vaultToken);
 
        // Prepare vault configuration
        const vaultConfig = {
            vaultAddr: secretsManager.vaultAddr,
            vaultNamespace: `${secretsManager.vaultNamespace}${tenantId}/${environmentType}`,
            vaultToken: vaultToken,
            mount: 'secret'
        };
 
        // Fetch both passwords in parallel for better performance
        const [tcxUserPassword, dcServerPassword] = await Promise.all([
            secretsManager.readVaultSecret({
                ...vaultConfig,
                secretPath: VAULT_PATHS.TCX_USER,
                fieldName: FIELD_NAMES.TCX_USER_PASSWORD
            }),
            secretsManager.readVaultSecret({
                ...vaultConfig,
                secretPath: VAULT_PATHS.DC_SERVER,
                fieldName: FIELD_NAMES.DC_SERVER_PASSWORD
            })
        ]);
 
        if (!tcxUserPassword || !dcServerPassword) {
            throw new VaultError("One or more passwords could not be retrieved from Vault");
        }
 
        return { tcxUserPassword, dcServerPassword };
 
    } catch (error) {
        console.error(`[SecretsService] Vault error:`, error);
        console.error(`[SecretsService] Error name: ${error.name}`);
        console.error(`[SecretsService] Error message: ${error.message}`);
        console.error(`[SecretsService] Error code: ${error.Code}`);
        console.error(`[SecretsService] Error type: ${error.Type}`);
        if (error.$metadata) {
            console.error(`[SecretsService] HTTP Status: ${error.$metadata.httpStatusCode}`);
            console.error(`[SecretsService] Request ID: ${error.$metadata.requestId}`);
        }
        console.error(`[SecretsService] Full error object:`, JSON.stringify(error, null, 2));
        throw new VaultError("Failed to retrieve secrets from Vault", error.message);
    }
};
 
/**
 * Store passwords in database
 * @param {string} envId - Environment ID
 * @param {Object} passwords - Password data
 * @returns {Promise<void>}
 */
const storePasswordsInDatabase = async (envId, passwords) => {
    try {
        await Environment.findByIdAndUpdate(envId, {
            tcx_user_password: passwords.tcxUserPassword,
            dc_server_password: passwords.dcServerPassword,
            passwordsCachedAt: new Date()
        });
        console.log(`[SecretsService] Passwords stored in database for environment ${envId}`);
    } catch (error) {
        console.error(`[SecretsService] Failed to store passwords in database:`, error);
        // We don't throw here as we still have the passwords to return
        // This is a performance optimization failure, not a critical error
    }
};
 
/**
 * Main service method to get passwords for an environment
 * @param {string} envId - Environment ID
 * @returns {Promise<Object>} Password response object
 * @throws {BadRequestError|NotFoundError|VaultError|ConfigurationError} Various service errors
 */
export const getPasswordsForEnvironment = async (envId) => {
    console.log(`[SecretsService] Processing password request for environment: ${envId}`);
 
    // Handle bypass mode early
    if (isInBypassMode()) {
        console.log(`[SecretsService] Bypass mode enabled, returning demo passwords`);
        return getDemoPasswords();
    }
 
    // Validate inputs
    validateEnvironmentId(envId);
    validateConfiguration();
 
    // Find and validate environment
    const environment = await findAndValidateEnvironment(envId);
 
    // Check if passwords are already stored in database
    if (hasStoredPasswords(environment)) {
        console.log(`[SecretsService] Returning stored passwords from database`);
        return createPasswordResponse(environment, environment);
    }
 
    // Fetch passwords from Vault
    console.log(`[SecretsService] No stored passwords found, fetching from Vault`);
    const vaultPasswords = await fetchPasswordsFromVault(environment);
 
    // Store passwords for future use
    await storePasswordsInDatabase(envId, vaultPasswords);
 
    // Return password response
    return createPasswordResponse(vaultPasswords, environment);
};
 
/**
 * Check if passwords exist for an environment without fetching them
 * @param {string} envId - Environment ID
 * @returns {Promise<boolean>} True if passwords exist in database
 * @throws {BadRequestError|NotFoundError} Service errors
 */
export const checkPasswordsExist = async (envId) => {
    console.log(`[SecretsService] Checking password existence for environment: ${envId}`);
 
    // Handle bypass mode early - passwords always "exist" in bypass mode
    if (isInBypassMode()) {
        console.log(`[SecretsService] Bypass mode enabled, passwords always exist`);
        return true;
    }
 
    // Validate inputs
    validateEnvironmentId(envId);
 
    // Find and validate environment
    const environment = await findAndValidateEnvironment(envId);
 
    // Check if passwords are stored in database
    const exist = hasStoredPasswords(environment);
    console.log(`[SecretsService] Password existence check result: ${exist}`);
    
    return exist;
};
 
export default {
    getPasswordsForEnvironment,
    checkPasswordsExist,
    NotFoundError,
    BadRequestError,
    VaultError,
    ConfigurationError
};
