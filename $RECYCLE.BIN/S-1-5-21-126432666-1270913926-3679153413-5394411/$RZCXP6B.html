/**
 * Acquire a lock for an environment to prevent concurrent updates
 * @param {string} environmentId - The environment ID to lock
 * @param {string} context - Context information about what's acquiring the lock
 * @returns {Object|null} - Returns { env, lockId } if lock acquired, null otherwise
 */
async function acquireEnvironmentLock(environmentId, context = '') {
  const lockId = new mongoose.Types.ObjectId().toString();
  const env = await Environment.findOneAndUpdate(
    { 
      _id: environmentId, 
      $or: [
        { processingLock: { $exists: false } }, 
        { processingLock: null }
      ] 
    },
    { 
      processingLock: lockId, 
      processingLockContext: context, 
      processingLockAt: new Date() 
    },
    { new: true }
  );
  return env ? { env, lockId } : null;
}
 
/**
 * Release the lock for an environment
 * @param {string} environmentId - The environment ID to unlock
 * @param {string} lockId - The lock ID that was acquired
 */
async function releaseEnvironmentLock(environmentId, lockId) {
  await Environment.updateOne(
    { _id: environmentId, processingLock: lockId },
    { $unset: { processingLock: "", processingLockContext: "", processingLockAt: "" } }
  );
}
